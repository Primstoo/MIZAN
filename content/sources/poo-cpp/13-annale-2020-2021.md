## Exercice 1 · La classe Polynome

**Énoncé (résumé).** Un polynôme a0 + a1x + … + anxⁿ est stocké par son degré (`int degre`) et un tuple de coefficients (`int *Tuple`), par exemple 7 + 2x − 9x² + 8x⁴ → (7, 2, −9, 0, 8). Écrire : 1) le constructeur `Polynome(int d, const int *T)` ; 2) la recopie ; 3) le destructeur ; 4) `Afficher()` ; 5) à 8) `operator+`, `*`, `/` (division entière), `%` ; 9) `operator==` ; 10) `operator=` ; 11) `evaluer(int x)` ; 12) `getDegre()`.

> [!attention] Le sujet écrit `Operator` avec une majuscule et `Plynome` : en C++, c'est `operator` et `Polynome`. Autre limite : avec des coefficients **entiers**, la division euclidienne n'est exacte que si le coefficient dominant du diviseur divise les coefficients rencontrés (c'est le cas avec x − 1). Le corrigé fait une division entière des coefficients et s'arrête sinon ; écris cette hypothèse sur ta copie.

```etapes Méthode
Constructeur :: `Tuple = new int[degre + 1]`, puis copie des d + 1 coefficients.
Recopie et = :: copie **profonde** (nouveau tableau) ; dans `=`, test d'auto-affectation et `delete[]` de l'ancien.
Addition :: degré max des deux, on additionne terme à terme (0 au-delà du degré).
Produit :: degré d1 + d2, `r[i+j] += a[i] * b[j]`.
Division :: tant que deg(reste) ≥ deg(diviseur), on retire c·x^k·diviseur, c = dominant(reste) / dominant(diviseur).
Évaluer :: schéma de Horner, du coefficient de plus haut degré vers a0.
```

```cpp
#include <iostream>
using namespace std;

class Polynome {
    int degre;
    int* Tuple;            // Tuple[i] = coefficient de x^i
    void normaliser() {    // enlève les zéros de tête (garde au moins une valeur)
        while (degre > 0 && Tuple[degre] == 0) degre--;
    }
public:
    Polynome(int d = 0, const int* T = nullptr) : degre(d) {           // 1)
        Tuple = new int[degre + 1];
        for (int i = 0; i <= degre; i++) Tuple[i] = T ? T[i] : 0;
        if (T) normaliser();   // sans tuple : polynôme nul de taille d (pour les calculs)
    }
    Polynome(const Polynome& p) : degre(p.degre) {                     // 2)
        Tuple = new int[degre + 1];
        for (int i = 0; i <= degre; i++) Tuple[i] = p.Tuple[i];
    }
    ~Polynome() { delete[] Tuple; }                                    // 3)
    void Afficher() const {                                            // 4)
        bool premier = true;
        for (int i = 0; i <= degre; i++) {
            if (Tuple[i] == 0 && degre > 0) continue;
            int c = Tuple[i];
            if (!premier) cout << (c < 0 ? " - " : " + ");
            else if (c < 0) cout << "-";
            cout << (c < 0 ? -c : c);
            if (i >= 1) cout << "x";
            if (i >= 2) cout << "^" << i;
            premier = false;
        }
        cout << endl;
    }
    Polynome operator+(const Polynome& p) const {                      // 5)
        int d = degre > p.degre ? degre : p.degre;
        Polynome r(d);
        for (int i = 0; i <= d; i++)
            r.Tuple[i] = (i <= degre ? Tuple[i] : 0) + (i <= p.degre ? p.Tuple[i] : 0);
        r.normaliser();
        return r;
    }
    Polynome operator*(const Polynome& p) const {                      // 6)
        Polynome r(degre + p.degre);
        for (int i = 0; i <= degre; i++)
            for (int j = 0; j <= p.degre; j++)
                r.Tuple[i + j] += Tuple[i] * p.Tuple[j];
        r.normaliser();
        return r;
    }
    // division euclidienne : *this = q * p + r, deg r < deg p
    void diviser(const Polynome& p, Polynome& q, Polynome& r) const {
        r = *this;
        q = Polynome(degre >= p.degre ? degre - p.degre : 0);
        while (r.degre >= p.degre && !(r.degre == 0 && r.Tuple[0] == 0)) {
            int k = r.degre - p.degre;
            int c = r.Tuple[r.degre] / p.Tuple[p.degre];   // division entière
            if (c == 0) break;                              // plus divisible dans Z
            q.Tuple[k] = c;
            for (int j = 0; j <= p.degre; j++) r.Tuple[j + k] -= c * p.Tuple[j];
            int avant = r.degre;
            r.normaliser();
            if (r.degre == avant) break;                    // reste non annulable
        }
        q.normaliser();
    }
    Polynome operator/(const Polynome& p) const { Polynome q, r; diviser(p, q, r); return q; } // 7)
    Polynome operator%(const Polynome& p) const { Polynome q, r; diviser(p, q, r); return r; } // 8)
    int operator==(const Polynome& p) const {                          // 9)
        if (degre != p.degre) return 0;
        for (int i = 0; i <= degre; i++) if (Tuple[i] != p.Tuple[i]) return 0;
        return 1;
    }
    Polynome& operator=(const Polynome& p) {                           // 10)
        if (this != &p) {
            delete[] Tuple;
            degre = p.degre;
            Tuple = new int[degre + 1];
            for (int i = 0; i <= degre; i++) Tuple[i] = p.Tuple[i];
        }
        return *this;
    }
    int evaluer(int x) const {                                         // 11) Horner
        int s = 0;
        for (int i = degre; i >= 0; i--) s = s * x + Tuple[i];
        return s;
    }
    int getDegre() const { return degre; }                             // 12)
};

int main() {
    int t1[] = {7, 2, -9, 0, 8};          // 7 + 2x - 9x^2 + 8x^4
    int t2[] = {-1, 1};                   // x - 1
    Polynome P(4, t1), Q(1, t2);
    cout << "P = "; P.Afficher();
    cout << "Q = "; Q.Afficher();
    cout << "P + Q = "; (P + Q).Afficher();
    cout << "P * Q = "; (P * Q).Afficher();
    cout << "P / Q = "; (P / Q).Afficher();
    cout << "P % Q = "; (P % Q).Afficher();
    cout << "P(1) = " << P.evaluer(1) << ", P(2) = " << P.evaluer(2) << endl;
    Polynome R = (P / Q) * Q + (P % Q);
    cout << "(P/Q)*Q + P%Q == P ? " << (R == P) << endl;
    cout << "degre de P : " << P.getDegre() << endl;
    return 0;
}
```

```sortie
P = 7 + 2x - 9x^2 + 8x^4
Q = -1 + 1x
P + Q = 6 + 3x - 9x^2 + 8x^4
P * Q = -7 + 5x + 11x^2 - 9x^3 - 8x^4 + 8x^5
P / Q = 1 - 1x + 8x^2 + 8x^3
P % Q = 8
P(1) = 8, P(2) = 103
(P/Q)*Q + P%Q == P ? 1
degre de P : 4
```

> [!retenir] Vérification gratuite à l'examen : le reste de la division par x − 1 vaut P(1). Ici P(1) = 7 + 2 − 9 + 8 = **8**, c'est bien `P % Q`.

```exercice Questions de l'exercice 1
? Écris le constructeur de recopie de `Polynome`.
= `Polynome(const Polynome& p) : degre(p.degre) { Tuple = new int[degre + 1]; for (int i = 0; i <= degre; i++) Tuple[i] = p.Tuple[i]; }` : nouveau tableau, puis copie des valeurs (copie profonde).

? Pourquoi `operator=` commence-t-il par `delete[] Tuple` (après le test `this != &p`) ?
= L'objet de gauche possède déjà un tableau, peut-être de taille différente : on le libère pour ne pas avoir de fuite, puis on en alloue un de la bonne taille.

? Calcule P(2) pour P = 7 + 2x − 9x² + 8x⁴ avec Horner.
= Coefficients de haut en bas : 8, 0, −9, 2, 7. s = 8 → 16 → 23 → 48 → **103**.
reponse: 103

? Quel est le degré de P × Q avec deg P = 4 et deg Q = 1 ?
= 4 + 1 = **5** (le produit des coefficients dominants 8 × 1 n'est pas nul).
reponse: 5
```

## Exercice 2 · Employe, Responsable, Commercial, Personnel

**Énoncé (résumé).** Un employé a un nom, un matricule et un indice salarial ; salaire = indiceSalarial × valeur, avec valeur = 10 pour tous. a) constructeur, b) `afficher`, c) `salaire`. Un `Responsable` (dérivé) a un titre et un tableau de subordonnés directs : d) constructeur, e) `afficher` redéfinie, f) `afficherSubordonnesDirects`. Un `Commercial` (dérivé) a un fixe et un intéressement sur ses ventes du mois : g) constructeur, h) `enregistreVentes`, i) `salaire` = indiceSalarial × valeur + venteDuMois / 10. Enfin une classe `personnel` qui calcule la somme des salaires à verser.

```etapes Méthode
valeur commune :: `static const int valeur = 10;` (un seul exemplaire pour la classe).
salaire et afficher :: `virtual` dans `Employe` : `Personnel` manipule des `Employe*`.
Subordonnés :: `Employe**` + compteur : un responsable peut diriger des commerciaux comme des employés.
Somme :: une boucle `s += tab[i]->salaire();` : la liaison dynamique choisit la bonne formule.
```

```cpp
#include <iostream>
#include <string>
using namespace std;

class Employe {
protected:
    string nom;
    int matricule;
    double indiceSalarial;
    static const int valeur = 10;          // la même pour tous les employés
public:
    Employe(string n, int m, double ind) : nom(n), matricule(m), indiceSalarial(ind) {} // a)
    virtual ~Employe() {}
    virtual void afficher() const {                                                     // b)
        cout << nom << " (matricule " << matricule << ", indice " << indiceSalarial << ")";
    }
    virtual double salaire() const { return indiceSalarial * valeur; }                  // c)
};

class Responsable : public Employe {
    string titre;
    Employe** subordonnes;     // inférieurs directs
    int nbSub, maxSub;
public:
    Responsable(string n, int m, double ind, string t, int max = 10)                    // d)
        : Employe(n, m, ind), titre(t), nbSub(0), maxSub(max) {
        subordonnes = new Employe*[maxSub];
    }
    ~Responsable() { delete[] subordonnes; }
    Responsable(const Responsable&) = delete;
    Responsable& operator=(const Responsable&) = delete;
    void ajouterSubordonne(Employe* e) { if (nbSub < maxSub) subordonnes[nbSub++] = e; }
    void afficher() const override {                                                    // e)
        Employe::afficher(); cout << ", " << titre;
    }
    void afficherSubordonnesDirects() const {                                           // f)
        cout << "Subordonnes de " << nom << " :" << endl;
        for (int i = 0; i < nbSub; i++) { cout << "  "; subordonnes[i]->afficher(); cout << endl; }
    }
};

class Commercial : public Employe {
    double venteDuMois;
public:
    Commercial(string n, int m, double ind, double v = 0) : Employe(n, m, ind), venteDuMois(v) {} // g)
    void enregistreVentes(double v) { venteDuMois = v; }                                // h)
    double salaire() const override { return indiceSalarial * valeur + venteDuMois / 10; } // i)
    void afficher() const override { Employe::afficher(); cout << ", commercial"; }
};

class Personnel {
    Employe** tab;
    int nb, max;
public:
    Personnel(int m = 100) : nb(0), max(m) { tab = new Employe*[max]; }
    ~Personnel() { delete[] tab; }
    Personnel(const Personnel&) = delete;
    Personnel& operator=(const Personnel&) = delete;
    void ajouter(Employe* e) { if (nb < max) tab[nb++] = e; }
    double masseSalariale() const {
        double s = 0;
        for (int i = 0; i < nb; i++) s += tab[i]->salaire();   // virtual : bon salaire
        return s;
    }
};

int main() {
    Employe e1("Amine", 1, 300);
    Commercial c1("Nora", 2, 250);
    c1.enregistreVentes(5000);
    Responsable r1("Youssef", 3, 500, "Chef de service");
    r1.ajouterSubordonne(&e1);
    r1.ajouterSubordonne(&c1);
    r1.afficher(); cout << endl;
    r1.afficherSubordonnesDirects();
    Personnel p;
    p.ajouter(&e1); p.ajouter(&c1); p.ajouter(&r1);
    cout << "Salaire de Nora : " << c1.salaire() << endl;
    cout << "Total a verser : " << p.masseSalariale() << endl;
    return 0;
}
```

```sortie
Youssef (matricule 3, indice 500), Chef de service
Subordonnes de Youssef :
  Amine (matricule 1, indice 300)
  Nora (matricule 2, indice 250), commercial
Salaire de Nora : 3000
Total a verser : 11000
```

> [!piege] Sans `virtual double salaire()`, `Personnel` calculerait le salaire de Nora comme un employé ordinaire (2 500 au lieu de 3 000), et le total serait faux sans aucune erreur de compilation.

```exercice Questions de l'exercice 2
? Pourquoi `valeur` est-elle `static` ?
= Elle est la même pour tous les employés : un seul exemplaire partagé par la classe, au lieu d'une copie par objet.

? Salaire de Nora : indice 250, ventes 5 000 ?
= 250 × 10 + 5 000 / 10 = 2 500 + 500 = **3 000**.
reponse: 3000

? Total versé pour Amine (indice 300), Nora (ci-dessus) et Youssef (responsable, indice 500) ?
= 3 000 + 3 000 + 5 000 = **11 000** (le responsable est payé comme un employé : `salaire` n'est pas redéfinie pour lui).
reponse: 11000
```

## Flashcards

```cartes
Classe avec int* + new :: destructeur, recopie profonde, operator= (règle des trois).
operator= :: `if (this != &p) { delete[] ancien; allouer; copier; } return *this;`
Produit de polynômes :: `r[i+j] += a[i] * b[j]`, degré d1 + d2.
Horner :: `s = s * x + a[i]` de i = n à 0.
Reste de P par (x − a) :: P(a).
Attribut commun à tous les objets :: `static`.
Calcul polymorphe sur un tableau :: tableau de pointeurs de base + méthode `virtual`.
Redéfinir en réutilisant la base :: `Employe::afficher();` puis le complément.
```

- 0
- 7

+ int
- void

+ est partagé par tous les employés
