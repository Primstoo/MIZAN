## 2016-2017 normale · Exercice 1 : point, cast et pointcol (1 h 30)

**Énoncé (résumé).** Classe `point` (x, y entiers) : 1) constructeur à 0, 1 ou 2 arguments ; 2) cast vers un entier (l'abscisse) ; 3)-4) `affiche` ; 5) `operator+` membre ; 6) que font `n = p1;`, `n = p1 + p2;`, `p = p1 + p2;`, `fct(p);` ? 7) `int operator==(point&, point&)` ami. 8)-9) `pointcol` dérivée avec une couleur `cl`, `affiche` redéfinie, `colore(int)`, constructeur avec valeurs par défaut. 10) Avec `a`, `b` de type `pointcol` et `p` de type `point`, sont-elles correctes : `a == b`, `a == p`, `p == a`, `1 == a` ?

```cpp
#include <iostream>
using namespace std;

class point {
protected:
    int x, y;
public:
    point(int a = 0, int b = 0) : x(a), y(b) {}             // 1) 0, 1 ou 2 arguments
    operator int() const { return x; }                       // 2) cast vers l'abscisse
    void affiche() const { cout << "(" << x << ", " << y << ")" << endl; } // 3) et 4)
    point operator+(const point& p) const { return point(x + p.x, y + p.y); } // 5)
    friend int operator==(const point& a, const point& b);   // 7) fonction amie
};
int operator==(const point& a, const point& b) { return a.x == b.x && a.y == b.y; }

class pointcol : public point {                              // 8)
    int cl;
public:
    pointcol(int a = 0, int b = 0, int c = 0) : point(a, b), cl(c) {} // 9)
    void affiche() const { cout << "(" << x << ", " << y << ") couleur " << cl << endl; }
    void colore(int c) { cl = c; }
};

void fct(int n) { cout << "fct recoit " << n << endl; }

int main() {
    point p, p1(1, 2), p2(3, 4);
    int n;
    n = p1;            cout << "instruction 1 : n = " << n << endl;
    n = p1 + p2;       cout << "instruction 2 : n = " << n << endl;
    p = p1 + p2;       cout << "instruction 3 : p = "; p.affiche();
    fct(p);            // instruction 4
    pointcol a(1, 2, 5), b(1, 2, 9);
    a.affiche();
    cout << "a == b : " << (a == b) << endl;   // i
    cout << "a == p : " << (a == p) << endl;   // ii
    cout << "p == a : " << (p == a) << endl;   // iii
    // (1 == a) : erreur de compilation, appel ambigu (voir question 10)
    return 0;
}
```

```sortie
instruction 1 : n = 1
instruction 2 : n = 4
instruction 3 : p = (4, 6)
fct recoit 4
(1, 2) couleur 5
a == b : 1
a == p : 0
p == a : 0
```

```exercice Questions 6 et 10
? Q6 : que font les quatre instructions ?
= 1) `n = p1` : cast `operator int`, n = 1. 2) `n = p1 + p2` : `operator+` donne le point (4, 6), puis cast : n = 4. 3) `p = p1 + p2` : somme de points, affectée à p. 4) `fct(p)` : p est converti en int (4) pour l'argument.

? Q10 : lesquelles des quatre comparaisons compilent ?
= `a == b`, `a == p`, `p == a` : oui, un `pointcol` est un `point` (conversion dérivée → base), seules x et y sont comparées. `1 == a` : **non**, l'appel est ambigu entre `operator==(point, point)` (1 converti en point) et le `==` des entiers (a converti en int).
```

## 2016-2017 normale · Exercice 2 : le vecteur dynamique vectd

**Énoncé (résumé).** `class vectd { int nelem; float *adr; ... }` : 1) constructeur `vectd(int = 3)` ; 2) destructeur ; 3) `affiche` au format < c1, c2, …, cn > ; 4) `vect_somme` ; 5) `pro_scalaire` amie ; 6) `operator=` ; 7) recopie ; 8) `operator[]` utilisable des deux côtés de `=` ; 9) programme de test ; 10) fonction `nombre` donnant le nombre de vecteurs existants.

```cpp
#include <iostream>
using namespace std;

class vectd {
    int nelem;
    float* adr;
    static int nb;                      // 10) nombre de vecteurs existants
public:
    vectd(int n = 3) : nelem(n) {       // 1)
        adr = new float[nelem];
        for (int i = 0; i < nelem; i++) adr[i] = 0;
        nb++;
    }
    ~vectd() { delete[] adr; nb--; }    // 2)
    vectd(const vectd& v) : nelem(v.nelem) {   // 7) recopie profonde
        adr = new float[nelem];
        for (int i = 0; i < nelem; i++) adr[i] = v.adr[i];
        nb++;
    }
    void affiche() const {              // 3) < c1, c2, ..., cn >
        cout << "< ";
        for (int i = 0; i < nelem; i++) cout << adr[i] << (i < nelem - 1 ? ", " : " >\n");
    }
    vectd vect_somme(const vectd& v) const {   // 4) (même dimension supposée)
        vectd r(nelem);
        for (int i = 0; i < nelem; i++) r.adr[i] = adr[i] + v.adr[i];
        return r;
    }
    friend float pro_scalaire(const vectd& a, const vectd& b);   // 5)
    float& operator[](int i) {          // 8) référence : utilisable à gauche de =
        if (i < 0 || i >= nelem) i = 0;
        return adr[i];
    }
    vectd& operator=(const vectd& v) {  // 6)
        if (this != &v) {
            delete[] adr;
            nelem = v.nelem;
            adr = new float[nelem];
            for (int i = 0; i < nelem; i++) adr[i] = v.adr[i];
        }
        return *this;
    }
    static int nombre() { return nb; }  // 10)
};
int vectd::nb = 0;

float pro_scalaire(const vectd& a, const vectd& b) {
    float s = 0;
    for (int i = 0; i < a.nelem; i++) s += a.adr[i] * b.adr[i];
    return s;
}

int main() {                            // 9)
    vectd u, v;
    for (int i = 0; i < 3; i++) { u[i] = i + 1; v[i] = 2 * i; }
    u.affiche(); v.affiche();
    vectd w = u.vect_somme(v);
    w.affiche();
    cout << "u.v = " << pro_scalaire(u, v) << endl;
    cout << "vecteurs existants : " << vectd::nombre() << endl;
    {
        vectd t(5);
        t = u;                          // affectation, puis t a 3 composantes
        t.affiche();
        cout << "vecteurs existants : " << vectd::nombre() << endl;
    }
    cout << "vecteurs existants : " << vectd::nombre() << endl;
    return 0;
}
```

```sortie
< 1, 2, 3 >
< 0, 2, 4 >
< 1, 4, 7 >
u.v = 16
vecteurs existants : 3
< 1, 2, 3 >
vecteurs existants : 4
vecteurs existants : 3
```

```exercice Questions clés
? Q8 : pourquoi `operator[]` renvoie-t-il `float&` ?
= Pour que `u[i] = 5;` modifie la composante : une référence est une lvalue. Avec `float` (par valeur), l'affectation porterait sur une copie temporaire et ne compilerait pas.

? Q10 : comment compter les vecteurs existants ?
= Un membre `static int nb` (défini hors classe : `int vectd::nb = 0;`), incrémenté dans **chaque** constructeur (y compris la recopie), décrémenté dans le destructeur, et une fonction `static int nombre()`.
```

## 2016-2017 rattrapage · complexe ↔ point, vect × matrice (1 h)

**Énoncé (résumé).** Exercice 1 : `complexe` (constructeur 0/1/2 args, `affiche` « reel + imag*i », `+`, `-`, cast vers `point`) et `point` (constructeur, `affiche`, cast vers `complexe`, `+`). Que font `p = c1 + c2;` et `c = p1 + p2;` ? Exercice 2 : classes `vect` (3 composantes) et `matrice` (3 × 3, construite depuis l'adresse d'un tableau), fonction indépendante `produit` matrice × vecteur, programme de test.

```cpp
#include <iostream>
using namespace std;

class complexe;   // déclaration anticipée

class point {
    double x, y;
public:
    point(double a = 0, double b = 0) : x(a), y(b) {}       // 6)
    void affiche() const { cout << "point (" << x << ", " << y << ")" << endl; } // 7)
    operator complexe() const;                              // 8) défini après complexe
    point operator+(const point& p) const { return point(x + p.x, y + p.y); } // 9)
};

class complexe {
    double reel, imag;
public:
    complexe(double r = 0, double i = 0) : reel(r), imag(i) {}  // 1)
    void affiche() const { cout << reel << " + " << imag << "*i" << endl; } // 2)
    complexe operator+(const complexe& c) const { return complexe(reel + c.reel, imag + c.imag); } // 3)
    complexe operator-(const complexe& c) const { return complexe(reel - c.reel, imag - c.imag); } // 4)
    operator point() const { return point(reel, imag); }        // 5)
};

point::operator complexe() const { return complexe(x, y); }

int main() {
    point p, p1(4, 2), p2(3, 1);
    complexe c, c1(0, 5), c2(8, 10);
    p = c1 + c2;     // instruction 1 : somme de complexes, puis cast en point
    p.affiche();
    c = p1 + p2;     // instruction 2 : somme de points, puis cast en complexe
    c.affiche();
    (c2 - c1).affiche();
    return 0;
}
```

```sortie
point (8, 15)
7 + 3*i
8 + 5*i
```

```cpp
#include <iostream>
using namespace std;

class matrice;

class vect {
    double v[3];
public:
    vect(double a = 0, double b = 0, double c = 0) { v[0] = a; v[1] = b; v[2] = c; }
    void affiche() const { cout << "(" << v[0] << ", " << v[1] << ", " << v[2] << ")" << endl; }
    friend vect produit(const matrice& m, const vect& x);
};

class matrice {
    double m[3][3];
public:
    matrice(double t[3][3]) {           // adresse d'un tableau 3 x 3
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++) m[i][j] = t[i][j];
    }
    friend vect produit(const matrice& m, const vect& x);
};

// fonction indépendante, amie des deux classes (membres privés)
vect produit(const matrice& a, const vect& x) {
    vect r;
    for (int i = 0; i < 3; i++) {
        r.v[i] = 0;
        for (int j = 0; j < 3; j++) r.v[i] += a.m[i][j] * x.v[j];
    }
    return r;
}

int main() {
    double t[3][3] = { {1, 2, 3}, {4, 5, 6}, {7, 8, 9} };
    matrice M(t);
    vect w(1, 1, 1);
    produit(M, w).affiche();
    produit(M, vect(1, 0, -1)).affiche();
    return 0;
}
```

```sortie
(6, 15, 24)
(-2, -2, -2)
```

> [!retenir] Deux classes qui se convertissent l'une en l'autre : déclare la seconde à l'avance (`class complexe;`) et définis l'opérateur de cast **après** les deux classes. Une fonction qui lit les membres privés de deux classes doit être **amie des deux**.

## 2017-2018 normale · complexe complet, Personnel et Patron (1 h 45)

**Énoncé (résumé).** Exercice 1 : classe `complexe` (Re, Im) en `Complexe.h`/`Complexe.cpp` : constructeur, `Affiche`, `GetRe`/`GetIm`, `GetMod`/`GetArg` (avec `sqrt`, `acos`, `asin`), puis `=`, `+`, `-`, moins unaire, `/`, `*`, `*=`, `==`, `!=`, `<<`, `>>`. Exercice 2 : `Personnel` (Nom, Position, Salaire) avec `Affiche`, et `Patron` dérivée (BonusAnnuel, MarqueVoiture) qui redéfinit `Affiche`.

```cpp
#include <iostream>
#include <cmath>
using namespace std;

class complexe {
    double Re, Im;
public:
    complexe(double r = 0, double i = 0) : Re(r), Im(i) {}             // 1)
    void Affiche() const { cout << Re << (Im < 0 ? " - " : " + ") << fabs(Im) << "i" << endl; } // 2)
    double GetRe() const { return Re; }                                // 3)
    double GetIm() const { return Im; }
    double GetMod() const { return sqrt(Re * Re + Im * Im); }          // 4)
    double GetArg() const {                                            // avec acos, comme suggéré
        double m = GetMod();
        if (m == 0) return 0;
        double a = acos(Re / m);
        return Im < 0 ? -a : a;                                        // signe donné par Im
    }
    complexe& operator=(const complexe& c) { Re = c.Re; Im = c.Im; return *this; } // 5)
    complexe operator+(const complexe& c) const { return complexe(Re + c.Re, Im + c.Im); } // 6)
    complexe operator-(const complexe& c) const { return complexe(Re - c.Re, Im - c.Im); }
    complexe operator-() const { return complexe(-Re, -Im); }          // 7) moins unaire
    complexe operator/(const complexe& c) const {                     // 8)
        double d = c.Re * c.Re + c.Im * c.Im;                          // |c|², supposé non nul
        return complexe((Re * c.Re + Im * c.Im) / d, (Im * c.Re - Re * c.Im) / d);
    }
    complexe operator*(const complexe& c) const {                     // 9)
        return complexe(Re * c.Re - Im * c.Im, Re * c.Im + Im * c.Re);
    }
    complexe& operator*=(const complexe& c) { *this = *this * c; return *this; } // 10)
    bool operator==(const complexe& c) const { return Re == c.Re && Im == c.Im; } // 11)
    bool operator!=(const complexe& c) const { return !(*this == c); }
    friend ostream& operator<<(ostream& os, const complexe& c) {      // 12)
        return os << "(" << c.Re << ", " << c.Im << ")";
    }
    friend istream& operator>>(istream& is, complexe& c) { return is >> c.Re >> c.Im; }
};

int main() {
    complexe v1(3, 4), v2(1, -2), v3;
    v3 = v1 + v2;  cout << "v1 + v2 = " << v3 << endl;
    v3 = v1 - v2;  cout << "v1 - v2 = " << v3 << endl;
    v3 = -v1;      cout << "-v1 = " << v3 << endl;
    v3 = v1 * v2;  cout << "v1 * v2 = "; v3.Affiche();
    v3 = v1 / v2;  cout << "v1 / v2 = "; v3.Affiche();
    v3 *= v2;      cout << "(v1 / v2) * v2 = " << v3 << endl;
    cout << "|v1| = " << v1.GetMod() << ", arg(v2) = " << v2.GetArg() << endl;
    cout << "v3 == v1 : " << (v3 == v1) << ", v1 != v2 : " << (v1 != v2) << endl;
    return 0;
}
```

```sortie
v1 + v2 = (4, 2)
v1 - v2 = (2, 6)
-v1 = (-3, -4)
v1 * v2 = 11 - 2i
v1 / v2 = -1 + 2i
(v1 / v2) * v2 = (3, 4)
|v1| = 5, arg(v2) = -1.10715
v3 == v1 : 1, v1 != v2 : 1
```

#include <iostream>

using namespace std;

class Personnel {

public:

    }

    Personnel e("Salma", "Technicienne", 6000);

```exercice Calculs à refaire à la main
? Calcule (3 + 4i) ÷ (1 − 2i).
= On multiplie par le conjugué : (3 + 4i)(1 + 2i) / (1 + 4) = (3 + 6i + 4i − 8) / 5 = (−5 + 10i) / 5 = **−1 + 2i**.

? Pourquoi `<<` et `>>` sont-ils des fonctions amies et pas des membres ?
= Leur opérande de gauche est un flux (`cout`, `cin`), pas un complexe. Un membre aurait `this` à gauche.

? Arg avec `acos` seul suffit-il ?
= Non : `acos` renvoie un angle dans [0, π]. Il faut prendre l'opposé quand Im < 0 (c'est ce que fait `GetArg`).
```

## 2017-2018 rattrapage · la classe rationnel (1 h)

**Énoncé (résumé).** `rationnel` (p, q entiers premiers entre eux) : constructeur 0/1/2 paramètres, `Affiche`, `=`, `+`, `-`, `*`, `/`, `+=`, `==`, moins unaire, `++` pré et post, conversion en `float`, `^(unsigned)` (x = 2/5 → x^4 = 16/625), `^(int)` (x^−4 = 625/16), `>>`, `<<`, programme de test, puis `PGCD(rationnel r)` amie pour garder la fraction réduite. Le même exercice est repris en 2018-2019 normale (exercice 2).

```cpp
#include <iostream>
using namespace std;

class rationnel {
    int p, q;                                   // q > 0, fraction toujours réduite
    void reduire() {
        if (q < 0) { p = -p; q = -q; }
        int g = PGCD(*this);
        if (g > 1) { p /= g; q /= g; }
    }
public:
    rationnel(int a = 0, int b = 1) : p(a), q(b) { reduire(); }   // 1) q ≠ 0 supposé
    void Affiche() const { cout << p; if (q != 1) cout << "/" << q; cout << endl; } // 2)
    rationnel& operator=(const rationnel& r) { p = r.p; q = r.q; return *this; }    // 3)
    rationnel operator+(const rationnel& r) const { return rationnel(p * r.q + r.p * q, q * r.q); } // 4)
    rationnel operator-(const rationnel& r) const { return rationnel(p * r.q - r.p * q, q * r.q); } // 5)
    rationnel operator*(const rationnel& r) const { return rationnel(p * r.p, q * r.q); }          // 6)
    rationnel operator/(const rationnel& r) const { return rationnel(p * r.q, q * r.p); }          // 7)
    rationnel& operator+=(const rationnel& r) { return *this = *this + r; }                       // 8)
    bool operator==(const rationnel& r) const { return p == r.p && q == r.q; } // 9) (formes réduites)
    rationnel operator-() const { return rationnel(-p, q); }                    // 10)
    rationnel& operator++() { p += q; return *this; }                           // 11) pré
    rationnel operator++(int) { rationnel avant = *this; p += q; return avant; } //     post
    operator float() const { return (float)p / q; }                             // 12)
    rationnel operator^(unsigned n) const {                                     // 13)
        rationnel r(1);
        for (unsigned i = 0; i < n; i++) r = r * *this;
        return r;
    }
    rationnel operator^(int n) const {                                          // 14)
        if (n >= 0) return *this ^ (unsigned)n;
        return rationnel(q, p) ^ (unsigned)(-n);
    }
    friend istream& operator>>(istream& is, rationnel& r) { is >> r.p >> r.q; r.reduire(); return is; } // 15)
    friend ostream& operator<<(ostream& os, const rationnel& r) { return os << r.p << "/" << r.q; }      // 16)
    friend int PGCD(rationnel r) {                                              // 18) Euclide
        int a = r.p < 0 ? -r.p : r.p, b = r.q;
        while (b != 0) { int t = a % b; a = b; b = t; }
        return a == 0 ? 1 : a;
    }
};

int main() {                                                                    // 17)
    rationnel x(4, 10), y(1, 3);
    cout << "x = " << x << ", y = " << y << endl;
    cout << "x + y = " << x + y << ", x - y = " << x - y << endl;
    cout << "x * y = " << x * y << ", x / y = " << x / y << endl;
    cout << "x ^ 4u = " << (x ^ 4u) << ", x ^ -4 = " << (x ^ -4) << endl;
    rationnel z = x++;
    cout << "z = x++ : z = " << z << ", x = " << x << endl;
    ++x; cout << "++x : " << x << endl;
    x += y; cout << "x += y : " << x << ", en float : " << (float)x << endl;
    cout << "-y = " << -y << ", 2/4 == 1/2 : " << (rationnel(2, 4) == rationnel(1, 2)) << endl;
    return 0;
}
```

```sortie
x = 2/5, y = 1/3
x + y = 11/15, x - y = 1/15
x * y = 2/15, x / y = 6/5
x ^ 4u = 16/625, x ^ -4 = 625/16

```
